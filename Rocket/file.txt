      import { useGLTF } from "@react-three/drei"
      import { useEffect } from "react"
      
      export default function Model() {
        const { scene } = useGLTF("./rocket1.glb")
      
        useEffect(() => {
          scene.traverse((child) => {
            if (child.isMesh) {
      
              // Default: Rocket casts shadow
              child.castShadow = true
      
              // If this part is the base, let it receive the shadow
              if (child.name === "rocketBase") {
                child.receiveShadow = true
              }
            }
          })
        }, [scene])
      
        return <primitive object={scene} scale={0.4} rotation={[0,10,0]} />
      }
